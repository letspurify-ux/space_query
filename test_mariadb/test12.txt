-- ==========================================================================
-- SPACE QUERY MARIADB SINGLE-STATEMENT FORMATTER FINAL BOSS IV
-- Target: MariaDB 12.2+ (live-certified on MariaDB 12.2.2)
-- One repeatable statement: recursive CTEs, JSON_TABLE, dynamic columns,
-- vectors, PERCENTILE_DISC, INTERSECT ALL/EXCEPT ALL, and deep JSON frames.
-- ==========================================================================
WITH RECURSIVE
params AS(
SELECT DATE'2026-03-01'AS start_day,6 AS day_count
),
customers(customer_id,parent_customer_id,customer_name,segment_code,active_flag,profile_vec)AS(
SELECT 1,NULL,'Aurora Services','ENTERPRISE',TRUE,VEC_FromText('[1,0,0]')UNION ALL
SELECT 2,1,'Bluebird Works','GROWTH',TRUE,VEC_FromText('[0.8,0.2,0]')UNION ALL
SELECT 3,1,'Cobalt Systems','GROWTH',TRUE,VEC_FromText('[0.2,0.7,0.1]')UNION ALL
SELECT 4,2,'Dormant Branch','STARTER',FALSE,VEC_FromText('[0,0,1]')
),
orders(order_id,customer_id,order_day,order_status,attribute_blob,payload)AS(
SELECT 101,1,DATE'2026-03-01','PAID',COLUMN_CREATE('priority',3,'region','APAC'),JSON_OBJECT('channel','web','items',JSON_ARRAY(
JSON_OBJECT('sku','A-RED','category','HARDWARE','qty',2,'price',25,'discount',0,'tags',JSON_ARRAY('priority','red')),
JSON_OBJECT('sku','B-BLUE','category','SERVICE','qty',1,'price',70,'discount',0,'tags',JSON_ARRAY('install'))
))UNION ALL
SELECT 102,1,DATE'2026-03-03','PENDING',COLUMN_CREATE('priority',2,'region','EMEA'),JSON_OBJECT('channel','partner','items',JSON_ARRAY(
JSON_OBJECT('sku','C-GREEN','category','SOFTWARE','qty',1,'price',20,'discount',0,'tags',JSON_ARRAY('trial')),
JSON_OBJECT('sku','D-GOLD','category','SERVICE','qty',1,'price',40,'discount',0,'tags',JSON_ARRAY('renewal','gold'))
))UNION ALL
SELECT 201,2,DATE'2026-03-02','PAID',COLUMN_CREATE('priority',3,'region','APAC'),JSON_OBJECT('channel','store','items',JSON_ARRAY(
JSON_OBJECT('sku','E-WHITE','category','HARDWARE','qty',2,'price',50,'discount',0,'tags',JSON_ARRAY('bundle')),
JSON_OBJECT('sku','F-BLACK','category','SOFTWARE','qty',1,'price',50,'discount',0,'tags',JSON_ARRAY('license'))
))UNION ALL
SELECT 202,2,DATE'2026-03-04','REFUNDED',COLUMN_CREATE('priority',1,'region','AMER'),JSON_OBJECT('channel','web','items',JSON_ARRAY(
JSON_OBJECT('sku','G-SILVER','category','SERVICE','qty',1,'price',35,'discount',0,'tags',JSON_ARRAY('refund'))
))UNION ALL
SELECT 301,3,DATE'2026-03-02','PAID',COLUMN_CREATE('priority',3,'region','APAC'),JSON_OBJECT('channel','direct','items',JSON_ARRAY(
JSON_OBJECT('sku','H-ORANGE','category','HARDWARE','qty',3,'price',40,'discount',0,'tags',JSON_ARRAY('bulk','orange')),
JSON_OBJECT('sku','I-PURPLE','category','SOFTWARE','qty',2,'price',40,'discount',0,'tags',JSON_ARRAY('analytics'))
))UNION ALL
SELECT 302,3,DATE'2026-03-05','PAID',COLUMN_CREATE('priority',2,'region','EMEA'),JSON_OBJECT('channel','partner','items',JSON_ARRAY(
JSON_OBJECT('sku','J-CYAN','category','SERVICE','qty',1,'price',50,'discount',0,'tags',JSON_ARRAY('support')),
JSON_OBJECT('sku','K-LIME','category','SOFTWARE','qty',1,'price',30,'discount',0,'tags',JSON_ARRAY('addon','lime'))
))
),
days(day_no,day_value)AS(
SELECT 0,start_day FROM params
UNION ALL
SELECT day_no+1,day_value+INTERVAL 1 DAY
FROM days JOIN params ON TRUE
WHERE day_no+1<day_count
),
org_paths(customer_id,parent_customer_id,customer_path,tree_depth)AS(
SELECT customer_id,parent_customer_id,CAST(customer_name AS CHAR(300)),0
FROM customers WHERE parent_customer_id IS NULL
UNION ALL
SELECT c.customer_id,c.parent_customer_id,CONCAT(p.customer_path,' / ',c.customer_name),p.tree_depth+1
FROM customers c JOIN org_paths p ON p.customer_id=c.parent_customer_id
),
line_items AS(
SELECT o.order_id,o.customer_id,o.order_day,o.order_status,
COLUMN_GET(o.attribute_blob,'priority'AS UNSIGNED)AS priority_no,
COLUMN_GET(o.attribute_blob,'region'AS CHAR)AS region_code,
j.line_no,j.sku,j.category_code,j.qty,j.unit_price,j.discount_rate,
ROUND(j.qty*j.unit_price*(1-j.discount_rate),2)AS net_amount
FROM orders o
JOIN JSON_TABLE(
o.payload,'$.items[*]'
COLUMNS(
line_no FOR ORDINALITY,
sku VARCHAR(30)PATH'$.sku',
category_code VARCHAR(30)PATH'$.category',
qty DECIMAL(10,2)PATH'$.qty',
unit_price DECIMAL(10,2)PATH'$.price',
discount_rate DECIMAL(10,4)PATH'$.discount'
)
)j ON TRUE
),
line_tags AS(
SELECT o.order_id,o.customer_id,j.line_no,j.tag_no,j.tag_value
FROM orders o
JOIN JSON_TABLE(
o.payload,'$.items[*]'
COLUMNS(
line_no FOR ORDINALITY,
NESTED PATH'$.tags[*]'
COLUMNS(tag_no FOR ORDINALITY,tag_value VARCHAR(30)PATH'$')
)
)j ON TRUE
),
order_facts AS(
SELECT order_id,customer_id,order_day,order_status,MAX(priority_no)AS priority_no,
MAX(region_code)AS region_code,SUM(net_amount)AS order_total,COUNT(*)AS line_count,
GROUP_CONCAT(sku ORDER BY line_no SEPARATOR',')AS sku_list
FROM line_items
GROUP BY order_id,customer_id,order_day,order_status
),
analytic_orders AS(
SELECT f.*,
ROW_NUMBER()OVER(PARTITION BY customer_id ORDER BY order_total DESC,order_id)AS amount_rank,
SUM(order_total)OVER(PARTITION BY customer_id ORDER BY order_day,order_id ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)AS running_total,
IFNULL(LAG(order_total,1)OVER(PARTITION BY customer_id ORDER BY order_day,order_id),0)AS previous_total,
PERCENTILE_DISC(0.5)WITHIN GROUP(ORDER BY order_total)OVER(PARTITION BY customer_id)AS median_order_total
FROM order_facts f
),
daily AS(
SELECT c.customer_id,d.day_value,IFNULL(SUM(f.order_total),0)AS day_amount,COUNT(f.order_id)AS order_count
FROM customers c CROSS JOIN days d
LEFT JOIN order_facts f ON f.customer_id=c.customer_id AND f.order_day=d.day_value
GROUP BY c.customer_id,d.day_value
),
daily_windows AS(
SELECT d.*,
SUM(day_amount)OVER(PARTITION BY customer_id ORDER BY day_value ROWS BETWEEN 2 PRECEDING AND CURRENT ROW)AS rolling_three_day,
IFNULL(LAG(day_amount,1)OVER(PARTITION BY customer_id ORDER BY day_value),0)AS previous_day_amount
FROM daily d
),
customer_rollup AS(
SELECT customer_id,COUNT(DISTINCT order_id)AS order_count,COUNT(*)AS line_count,
SUM(net_amount)AS gross_amount,ROUND(AVG(net_amount),2)AS average_line_amount,
JSON_ARRAYAGG(sku ORDER BY order_id,line_no)AS sku_json,
JSON_OBJECTAGG(sku,JSON_OBJECT('amount',net_amount,'category',category_code,'priority',priority_no))AS item_map
FROM line_items
GROUP BY customer_id
),
eligible(customer_id)AS(
(SELECT customer_id FROM customers WHERE active_flag=TRUE)
INTERSECT ALL
(SELECT customer_id FROM order_facts WHERE order_status='PAID')
EXCEPT ALL
(SELECT customer_id FROM order_facts WHERE order_total<0)
),
final_rows AS(
SELECT c.customer_id,c.customer_name,c.segment_code,c.active_flag,
p.customer_path,p.tree_depth,
IFNULL(r.order_count,0)AS order_count,IFNULL(r.line_count,0)AS line_count,
IFNULL(r.gross_amount,0)AS gross_amount,IFNULL(r.average_line_amount,0)AS average_line_amount,
r.sku_json,r.item_map,
CASE WHEN e.customer_id IS NULL THEN'INELIGIBLE'ELSE'ELIGIBLE'END AS eligibility,
(SELECT li.sku FROM line_items li WHERE li.customer_id=c.customer_id ORDER BY li.net_amount DESC,li.order_id,li.line_no LIMIT 1)AS top_sku,
(SELECT MAX(li.net_amount)FROM line_items li WHERE li.customer_id=c.customer_id)AS top_amount,
(SELECT JSON_ARRAYAGG(lt.tag_value ORDER BY lt.order_id,lt.line_no,lt.tag_no)FROM line_tags lt WHERE lt.customer_id=c.customer_id)AS tags_json,
(SELECT MAX(dw.rolling_three_day)FROM daily_windows dw WHERE dw.customer_id=c.customer_id)AS peak_three_day,
VEC_DISTANCE_EUCLIDEAN(c.profile_vec,VEC_FromText('[1,0,0]'))AS distance_from_enterprise,
NATURAL_SORT_KEY(c.customer_name)AS natural_customer_key
FROM customers c
JOIN org_paths p ON p.customer_id=c.customer_id
LEFT JOIN customer_rollup r ON r.customer_id=c.customer_id
LEFT JOIN(SELECT DISTINCT customer_id FROM eligible)AS e ON e.customer_id=c.customer_id
)
SELECT
CASE
WHEN(SELECT COUNT(*)FROM orders)=6
AND(SELECT COUNT(*)FROM line_items)=11
AND(SELECT SUM(order_total)FROM order_facts)=645
AND(SELECT COUNT(DISTINCT customer_id)FROM eligible)=3
AND(SELECT COUNT(*)FROM org_paths)=4
AND(SELECT ROUND(MAX(median_order_total),0)FROM analytic_orders)=80
THEN'PASS'ELSE'FAIL'END AS status,
f.customer_id,f.customer_name,f.segment_code,f.active_flag,f.tree_depth,
f.order_count,f.line_count,f.gross_amount,f.average_line_amount,
f.eligibility,f.top_sku,f.top_amount,f.peak_three_day,
ROUND(f.distance_from_enterprise,6)AS distance_from_enterprise,
HEX(f.natural_customer_key)AS natural_customer_key_hex,
JSON_NORMALIZE(JSON_OBJECT(
'path',f.customer_path,
'skus',f.sku_json,
'tags',f.tags_json,
'items',f.item_map,
'top',JSON_OBJECT('sku',f.top_sku,'amount',f.top_amount),
'metrics',JSON_OBJECT(
'gross',f.gross_amount,
'distance',ROUND(f.distance_from_enterprise,6),
'all_tags',(SELECT COUNT(*)FROM line_tags)
)
))AS evidence
FROM final_rows f
ORDER BY f.customer_id;
